# Allow selected traditional search engines; block all other crawlers
# search engines selected based on their reach: global and geos/langs
# of interest; see table at end of file
# 
# Inform each named crawler they are allowed to crawl all of the site
# except contents of /assets dir and certain files in root dir
# The empty line at the end of for loop is intentional
#
# The default rule (must be at the end as shown) directs crawlers to
# access only the files that describe site terms. The disallow directive
# on dir /wip is honeypot: together with file /wip/index.html, it helps
# identify and understand patterns of errant bots.
#
# NB: 
#   1. crawler/bot name in a rule is case-sensitive
#   2. Internet Archive bots are known to ignore robots.txt but we list
#      them to document they were permitted to crawl
#   3. We allow archive bots in /assets so they can render the site fully
#
# TODO: establish need for sitemap; create if beneficial
#
layout: none
permalink: /robots.txt

{%- assign empty_line = "
" -%}

{%- comment -%} make a rule for each named crawler; write empty line after each rule {%- endcomment -%}
{%- for crawler in crawlers -%}
User-agent: {{ crawler | strip }}
Disallow: /.nojekyll
Disallow: /CNAME
Disallow: /*.md
{%- unless crawler == "ia_archiver" or crawler == "archive.org_bot" -%}
Disallow: /assets/
{%- endunless -%}
{{ empty_line }}
{%- endfor -%}

{%- comment -%} default rule {%- endcomment -%}
User-agent: *
Allow: /.well-known/tdmrep.json
Allow: /tdm-policy.json
Allow: /eng/terms.html
Disallow: /wip/
Disallow: /
